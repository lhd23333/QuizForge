% pandoc 试卷 LaTeX 模板（ctex 中文 + amsmath 数学）
% 用法：pandoc 试卷.md -o 试卷.tex --template templates/exam_template.tex
% 编译：xelatex 试卷.tex
% fontset=fandol：锁定 Fandol 开源字体（随 MiKTeX/TeX Live 分发，可自动安装），
% 不用系统默认字体（Windows 上是宋体/黑体等系统字体）。原因：本项目要多机部署
% （config.py 里 PANDOC/XELATEX 路径注释已提到"部署到其他机器"），不锁字体的话
% 换一台机器渲染出来的字宽、行高都会跟着系统字体变，PDF 效果不可控。
\documentclass[11pt,fontset=fandol]{ctexart}
% Fandol 的 CJK 子集在 xdvipdfmx 产物里没有 /ToUnicode，PDFium 能画出来，
% 但部分 Electron/PDF.js 阅读器会把中文渲染成空白。所有导出模式统一优先改用带完整
% Unicode 映射的 Noto Sans SC；Windows 没装 Noto 时回落到系统自带的微软雅黑。
% 两者都不存在时保留 ctex 已加载的 Fandol，至少不让导出直接失败。
$if(cjk_font)$
% 导出面板显式选定的中文字体（pandoc -V cjk_font，值经后端白名单校验）。
% 字体被卸载等边缘情况：探测失败即留空，不报错、不中断导出。
\IfFontExistsTF{$cjk_font$}{%
  \IfFontExistsTF{$cjk_font$ Bold}{%
    \setCJKmainfont{$cjk_font$}[BoldFont={$cjk_font$ Bold}]%
    \setCJKsansfont{$cjk_font$}[BoldFont={$cjk_font$ Bold}]%
  }{%
    \setCJKmainfont{$cjk_font$}%
    \setCJKsansfont{$cjk_font$}%
  }%
}{}%
$else$
\IfFontExistsTF{Noto Sans SC}{%
  \IfFontExistsTF{Noto Sans SC Bold}{%
    \setCJKmainfont{Noto Sans SC}[BoldFont={Noto Sans SC Bold}]%
    \setCJKsansfont{Noto Sans SC}[BoldFont={Noto Sans SC Bold}]%
  }{%
    % 「家族内含 Bold」的安装形态（可变字体/单文件多字重，Windows 字体库里
    % 只有 "Noto Sans SC" 一条）：独立 Bold 名不存在，显式指定会把 XeTeX
    % 直接打崩（fontspec Error: font "Noto Sans SC Bold" cannot be found
    % → No pages of output；2026-10-04 真实导出故障）。此时省略 BoldFont，
    % 让 fontspec 按「家族+字重」自行解析；解析不到也只会警告退化，不中断。
    \setCJKmainfont{Noto Sans SC}%
    \setCJKsansfont{Noto Sans SC}%
  }%
}{%
  \IfFontExistsTF{Microsoft YaHei}{%
    \setCJKmainfont{Microsoft YaHei}[BoldFont={Microsoft YaHei Bold}]%
    \setCJKsansfont{Microsoft YaHei}[BoldFont={Microsoft YaHei Bold}]%
  }{}%
}
$endif$
$if(latin_font)$
% 正文西文字体（正文西文与数字；数学区字体不受影响）
\IfFontExistsTF{$latin_font$}{%
  \setmainfont{$latin_font$}%
  \setsansfont{$latin_font$}%
}{}%
$endif$
% 罗马数字、圈号及部分 Unicode 数学符号默认会被 xeCJK 判给西文字体；Latin Modern
% 不含这些字形，只报 Missing character 而继续出 PDF，结果是正文悄悄少字。明确归入
% CJK 字符类，让文本区的这些字符走 Noto/微软雅黑；数学区另由 exporter 转成
% 标准 LaTeX 命令，仍使用数学字体。
\xeCJKDeclareCharClass{CJK}{
  "03B1,"03B2,"03B3,"03B7,"03B8,"03BB,"03BE,"03C0,
  "2032,"2160 -> "217F,"2208,"2220,"2260,"2295,"2299,
  "2460 -> "2473,"25B3,"2A7D,"2A7E,"3001,"FF0E
}
\usepackage{amsmath,amssymb,amsfonts}
\usepackage{esint}      % \oiint 闭曲面积分（amssymb 不含）；此版本无 \oiiint，下面补
% 该 TeX Live 的 esint 只给 \oiint，不含 \oiiint（闭合体三重积分）。
% 本文件是 pandoc 模板，注释里也不能出现裸美元符号（会被当模板变量解析报错）。
% 故用纯命令写法：在闭曲面积分符号左侧再叠一个积分号，视觉上即三重闭合积分。
\providecommand{\oiiint}{\intop\!\!\!\oiint}
% —— 分数保持「外显」，靠行距吸收高度（QuizForge 排版）——
% 现象：导入管线给**每一条**行内公式都加了 \displaystyle（mechfix.add_displaystyle），
% 分数因此是行间尺寸，一行里几个分数就把行距顶开，题干和选项看着松散。
%
% **别把分数改小来治这个。** 试过一版把 frac 家族重定义成 \genfrac{}{}{}1
% （textstyle），确实不顶行了，但排出来的卷子不像卷子——真题的做法正相反：
% 查了语料里 2018–2024 全国卷/新课标真题合集，2024 新课标 II 第 8 题选项
% A. 1/8 B. 1/4 C. 1/2、2015 全国 II 文 12 选项 (1/3,1) 与 (-∞,-1/3)∪(1/3,+∞)、
% 同页题干 1/(1+x^2) 和 <= 2/3，**全部是 display 尺寸的大分数**。真卷里
% 题干与选项不分大小，也没有「选项用小分数」这一说。
%
% 真卷不显得挤，靠的是**行距**而不是小字号：选项行的行距明显宽于正文，
% 分数撑开的高度被行距吸收掉。所以这里也照办——公式尺寸一律不动，
% 只把选项行距（\settasks 的 after-item-skip，见下方 tasks 配置）放宽。
%
% 另两条路也别再试：去掉 \displaystyle 会把 \sum \int \lim 的上下限一起打回
% 算符右侧（实测 KaTeX 的 op-limits 从 1 掉到 0），把该外显的也毁了；把 \dfrac
% 改回 \frac 在 \displaystyle 作用域内**渲染完全一致**，可视输出逐字节相同，
% 只差 MathML 注解里回存的原始 TeX 串，纯属无效改动。

% 比例类（×\textheight，纯数字宏供 \setlength 拼接）：

% Python 侧对应：exporter._HALF_INLINE_HCAP(0.15)、_split_img_hcap(0.72 系数、

% QuizForge 题目插图统一版式：整题后靠右单独成行、等比缩放、
% 宽<=3.6cm 且 高<=3cm（keepaspectratio 保证哪边先顶到就按哪边缩，绝不拉伸变形）。
% exporter._stage_images 把 ![](file) 改写为 \qfig{file} 前的 md 图片引用，
% 由 _q_md/_solution_md 用本宏包裹。缺定义则 fallback 到普通 includegraphics。
% 参数：#1=图片文件名，#2=用户手动填写的图下说明（空则不显示）。
\newcommand{\qfig}[2]{%
  \par\nobreak\vspace{0.3em}%
  \noindent\hfill\begin{minipage}{3.6cm}%
    \centering\includegraphics[width=3.6cm,height=\qhcapfig,keepaspectratio]{#1}%
    \ifx\relax#2\relax\else\par\vspace{0.2em}{\footnotesize #2}\fi%
  \end{minipage}%
  \par\vspace{0.3em}%
}
% 填空题专用：左文右图。图浮在右侧、正文自动绕排（wrapfigure）。
% 宽度按图尺寸预留 4cm 栏（含间距），图本体等比缩放到 <=3.6cm x 3cm。
% 必须放在段首（wrapfig 要求），故 exporter 把它插在题干正文最前。
% 参数：#1=图片文件名，#2=图下说明文字。
\newcommand{\qfigwrap}[2]{%
  \begin{wrapfigure}{r}{4cm}%
    \vspace{-\baselineskip}%
    \centering\includegraphics[width=3.6cm,height=\qhcapfig,keepaspectratio]{#1}%
    \ifx\relax#2\relax\else\par\vspace{0.2em}{\footnotesize #2}\fi%
  \end{wrapfigure}%
}
% 双栏刷题的环绕图必须在本题结束前收口：短题干若尚未消耗完 wrapfig 预留行数，
% 下一道大题标题会继续绕上一题的图，甚至与图重叠。按宏包记录的剩余行数一次算出
% 应补高度，再调用公开的 \WFclear 恢复普通段落；不能逐行生成空段——图片多时会让
% XeLaTeX 反复重算 parshape，显著拖慢导出。长题干已填满时补高为 0，不制造留白。
% 该宏只由 exporter 的双栏环绕分支调用。
\makeatletter
\newcommand{\qwrapclear}{%
  \par
  \ifnum\c@WF@wrappedlines>\@ne
    \@tempcnta=\c@WF@wrappedlines\advance\@tempcnta by -\@ne
    \@tempdima=\baselineskip\multiply\@tempdima by \@tempcnta
    \vspace*{\@tempdima}%
  \fi
  \WFclear
}
% 环绕题开头至少预留一张图的高度；剩余栏高不足时让 TeX 在题前断栏。wrapfigure
% 本身是叠放盒，samepage 看不到它的真实高度，若不补这层提示，栏底短题可能把最后
% 一个小问甩到下一栏顶部。写成伸缩胶 + penalty 的 needspace 经典形式，不引入宏包。
\newcommand{\qwrapneed}[1]{%
  \par\begingroup
  \@tempdima=#1\relax
  \vskip\z@\@plus\@tempdima
  \penalty-100\vskip\z@\@plus-\@tempdima
  \vskip\@tempdima
  \penalty9999\vskip-\@tempdima
  \vskip\z@skip
  \endgroup
}

% 图片高度兜底上限。做成长度寄存器而不是写死在 \qfigflexbox 里，是为了让半页块
% （exporter._HALF_OPEN）能在自己的 TeX group 内临时压低它：那里正文高度锁死
% 0.5\textheight，按全页上限排的原位图会直接撑破版面。group 结束自动恢复，
% 故只有半页块受影响，其它模式的图片尺寸与改动前逐字一致。
\newlength{\qfigmaxh}
% 默认值在 geometry 声明之后设置（见下方「\qfigmaxh 默认限高求值点」）：
% 长度寄存器一次冻结，若在此处用 \textheight，拿到的是文档类默认版心
% （ctexart 约 541pt）而非真实版心，原位图上限会偏小约 30%（2026-10-04 P1-6）。
% 用户自定义排版的图片本体：仅负责按宽度占比缩放，左右对齐由 exporter 用
% \hfill 在外层拼好（见 exporter._place_image），避免在 TeX 宏里做字符串比较分支。
% 高度不设硬上限（keepaspectratio 保证不变形），但用 \qfigmaxh 兜底，
% 避免超大原图把单页撑爆。
% 参数：#1=宽度占比（0~1 小数，相对 \linewidth），#2=图片文件名，#3=图下说明文字。
\newcommand{\qfigflexbox}[3]{%
  \begin{minipage}{#1\linewidth}%
    \centering\includegraphics[width=\linewidth,height=\qfigmaxh,keepaspectratio]{#2}%
    \ifx\relax#3\relax\else\par\vspace{0.2em}{\footnotesize #3}\fi%
  \end{minipage}%
}
% 四图配选项（一个选项配一张图）的单格：选项标签在左、图片在右。
% 不能用 tasks 环境排——\task 里放不了 \includegraphics，这是这套网格另写的唯一
% 原因（见 exporter._pair_grid_latex）。列数与每格占比都在 Python 侧算好，
% 这里只负责一格的排版，格间的 \hfill / 换行同样由 exporter 拼。
% 外层和左右两个 minipage 都用 [t] 顶对齐，四张图高度不同也不会让 A-D 错落。
% 参数：#1=本格宽度占比（0~1，相对 \linewidth），#2=选项标签+文字，
%       #3=图片文件名，#4=图下说明文字。
\newcommand{\qpairitem}[4]{%
  \begin{minipage}[t]{#1\linewidth}%
    \setlength{\parskip}{0pt}\vspace{0pt}%
    \begin{minipage}[t]{1.8em}\vspace{0pt}\raggedright #2\end{minipage}\hfill%
    \begin{minipage}[t]{\dimexpr\linewidth-2.2em\relax}\vspace{0pt}%
      \centering\includegraphics[width=\linewidth,height=\qhcappair,keepaspectratio]{#3}%
      \ifx\relax#4\relax\else\par\vspace{0.2em}{\footnotesize #4}\fi%
    \end{minipage}%
  \end{minipage}%
}

% —— 自适应槽位（QuizForge 分页：1/4 页、半页、整页）——
% 把一题排进固定高度的槽位，内容超出目标槽位时**自动升级**到更大的槽位
% （1/4 → 半页 → 整页），超过整页则裸排、不截断内容。
%
% 为什么必须在 TeX 侧做：Python 侧（exporter.paginate）拿不到真实排版高度——
% 公式尺寸、图片、选项列数都影响高度，靠字数估算必然错判。这里先把内容排进
% lrbox 量出实际高度（\ht+\dp），再决定套多大的槽位。
%
% 断页判定也在这里：槽位高度超过本页余量就 \clearpage。所以「填空题结束后余量
% 够就放一道解答题、不够就另起一页顶头」不需要 Python 预先算页——那同样是算不准的。
\newsavebox{\qslotbox}
\newsavebox{\qslotsplit}   % 超长槽按页切出的临时段（见 \qslotclose）
\newlength{\qslotnat}
\newlength{\qslotgoal}
\newlength{\qslotfree}
\newlength{\qslotunit}
\newlength{\qslotused}
\newcommand{\qslotfrac}{1}
% 本页剩余高度。空页时 \pagegoal 仍是 \maxdimen（还没有 box 触发 output 例程），
% 此时余量就是整个版心；直接相减会得到荒谬的负值，故必须分支。
\newcommand{\qslotmeasurefree}{%
  \ifdim\pagegoal=\maxdimen
    \setlength{\qslotfree}{\textheight}%
  \else
    \setlength{\qslotfree}{\dimexpr\pagegoal-\pagetotal\relax}%
  \fi
}
% 两类规则的基准高度不同，用 \ifqslotpagerel 切换（exporter 按 mode 决定）：
%
% ① 相对模式（讲义/讲解/笔记：「一页四题」「一页两题」）——基准取**本页真实可用
%    高度**。不能直接用 \textheight：首页有标题块，余量小于版心，按 \textheight 切
%    四个 1/4 槽必然溢出、只放得下三题，「一页四题」就废了。
% ② 绝对模式（简单试卷：「每道解答题占半页纸」）——基准恒为 \textheight。这里的
%    「半页」是半张纸的**作答空间**，不能因为本页余量小就等比压缩，否则学生写不下；
%    余量不够就整题挪到下一页顶头（正是用户要的「另起一页顶头放下一道」）。
\newif\ifqslotpagerel
\qslotpagerelfalse
\newcommand{\qslotpageunit}{%
  \qslotmeasurefree
  \ifqslotpagerel
    \ifdim\pagegoal=\maxdimen
      \setlength{\qslotunit}{\textheight}%
      \setlength{\qslotused}{0pt}%
    \else
      \ifdim\qslotunit=0pt
        \setlength{\qslotunit}{\qslotfree}%
        \setlength{\qslotused}{0pt}%
      \fi
    \fi
  \else
    \setlength{\qslotunit}{\textheight}%
    \setlength{\qslotused}{\dimexpr\qslotunit-\qslotfree\relax}%
  \fi
}
% 目标槽位高度 = 比例 × 本页基准；量出的自然高度超了就逐级升级（1/4 → 半 → 整）。
\newcommand{\qslotsetgoal}{%
  \setlength{\qslotgoal}{\qslotfrac\qslotunit}%
  \ifdim\qslotnat>\qslotgoal
    \ifdim\qslotnat>0.5\qslotunit
      \setlength{\qslotgoal}{\qslotunit}%
    \else
      \setlength{\qslotgoal}{0.5\qslotunit}%
    \fi
  \fi
}
% #1 = 目标槽位占本页可用高度的比例（0.25 / 0.5 / 1）
%
% 收集为**可切的垂直材料**（vbox）而非 lrbox+minipage：minipage/lrbox 是
% 不可断的整体盒，超长题只能画到纸外（2026-10-04 审查实测内容溢出到
% y=-1599pt、无报错）；vbox 的段落可被 \qslotclose 用 \vsplit 按页切开。
% 限宽三件套与 \qpracticesolve 环境同款；正常槽位的外观与 minipage 版等价
% （同 \hsize、同字体、同 parskip）。
\newcommand{\qslotopen}[1]{%
  \edef\qslotfrac{#1}%
  \setbox\qslotbox=\vbox\bgroup
    \hsize=\linewidth
    \linewidth=\hsize
    \columnwidth=\hsize
    \setlength{\parskip}{0.4em}%
}
% \vss 吸收槽内底部余白，内容顶对齐；\nointerlineskip 防止基线间距把两个
% 半页槽的高度之和顶过 \textheight（那会把第二题挤到下一页）。
\newcommand{\qslotemit}[1]{%
  \par\nointerlineskip
  \vbox to #1{\vskip0pt\noindent\usebox{\qslotbox}\vss}%
  \par\nobreak
}
\newcommand{\qslotclose}{%
  \par\egroup
  \setlength{\qslotnat}{\dimexpr\ht\qslotbox+\dp\qslotbox\relax}%
  \qslotpageunit
  \qslotsetgoal
  \ifqslotpagerel
    \ifdim\qslotnat>\qslotunit
      % 相对模式基准=本页余量：超基准未必超整页——换到新页按整页基准重排，
      % 不提前独占（2026-10-04 审查实测：旧行为把它当「超整页」强制 clearpage，
      % 首页稍长的题白白废掉半页纸）。真超整页（nat>\textheight）留给下方切割。
      \ifdim\qslotnat>\textheight\else
        \clearpage
        \setlength{\qslotunit}{\textheight}%
        \setlength{\qslotused}{0pt}%
        \qslotsetgoal
      \fi
    \fi
  \fi
  \ifdim\qslotnat>\qslotunit
    % 超过整页：把槽内容按页切开、逐段放出——vbox 的段落可用 \vsplit 在页间
    % 断开（2026-10-04 审查：旧版 \usebox 放不可断整盒，实测内容溢出纸外
    % 且无报错）。非末段满页后强制换页，超长题按规则独占版面。
    \ifdim\qslotused>0pt \clearpage \fi
    \loop
      \setbox\qslotsplit=\vsplit\qslotbox to \textheight
      \ifvoid\qslotsplit\else
        \par\noindent\box\qslotsplit
        \ifvoid\qslotbox\else\clearpage\fi
      \fi
    \ifvoid\qslotbox\else\repeat
    \clearpage
    \setlength{\qslotunit}{\textheight}%
    \setlength{\qslotused}{0pt}%
  \else
    % 本页放不下这个槽位就换页，换页后基准与累计量重置、目标高度按新基准重算
    \ifdim\dimexpr\qslotused+\qslotgoal\relax>\qslotunit
      \clearpage
      \setlength{\qslotunit}{\textheight}%
      \setlength{\qslotused}{0pt}%
      \qslotsetgoal
    \fi
    \qslotemit{\qslotgoal}%
    \addtolength{\qslotused}{\qslotgoal}%
  \fi
}
